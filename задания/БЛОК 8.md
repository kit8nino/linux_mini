## Задание 14. Комплексный мини-проект (7.1–7.14)

Настройте своё окружение и автоматизируйте рутину.

1. Сделайте бэкап конфигов:
   ```bash
   cp ~/.bashrc ~/.bashrc.bak.$(date +%F)
   ```
2. Добавьте в `~/.bashrc`:
   ```bash
   cat >> ~/.bashrc << 'EOF'
   # === Настройки истории ===
   export HISTSIZE=10000
   export HISTFILESIZE=20000
   export HISTCONTROL=ignoreboth
   export HISTTIMEFORMAT="%F %T "

   # === Алиасы ===
   alias ll='ls -alh'
   alias ..='cd ..'
   alias ...='cd ../..'
   alias psg='ps aux | grep -v grep | grep -i'
   alias myip='curl -s ifconfig.me'

   # === Функции ===
   mkcd() { mkdir -p "$1" && cd "$1"; }
   EOF
   source ~/.bashrc
   ```
3. Создайте `~/bin/env_setup.sh`:
   ```bash
   mkdir -p ~/bin
   cat > ~/bin/env_setup.sh << 'EOF'
   #!/bin/bash
   set -euo pipefail

   PACKAGES=(htop tmux git curl jq)
   for pkg in "${PACKAGES[@]}"; do
       if ! command -v "$pkg" > /dev/null; then
           echo "Установить $pkg? (y/n)"
           read -r ans
           [ "$ans" = "y" ] && sudo apt install -y "$pkg"
       else
           echo "$pkg уже установлен"
       fi
   done

   if [ ! -f ~/.ssh/id_rsa ]; then
       echo "Генерирую SSH-ключ..."
       ssh-keygen -t rsa -b 4096 -N "" -f ~/.ssh/id_rsa
   fi

   git config --global user.name "Ваше Имя"
   git config --global user.email "you@example.com"
   echo "Готово"
   EOF
   chmod +x ~/bin/env_setup.sh
   ~/bin/env_setup.sh
   ```
4. Проверьте всё:
   ```bash
   ll /tmp | head -3
   mkcd /tmp/block8/test && pwd
   history | tail -5
   ```
5. Уберите за собой:
   ```bash
   cd /
   rm -rf /tmp/block8
   # При необходимости восстановите ~/.bashrc
   # cp ~/.bashrc.bak.$(date +%F) ~/.bashrc
   ```
